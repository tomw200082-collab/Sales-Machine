## 10. שאילתות לשחזור (שיטות, ⊥ עובדות)

```
-- הכנסות ולקוחות, ShopifyQL
FROM sales SHOW total_sales, orders, customers, new_customers, returning_customers TIMESERIES month SINCE -26M UNTIL today
FROM sales SHOW gross_sales, discounts, sales_reversals, total_sales, orders TIMESERIES month SINCE -8M UNTIL today
FROM sales SHOW total_sales, orders GROUP BY customer_name SINCE -365d UNTIL today ORDER BY total_sales DESC LIMIT 40
FROM sales SHOW orders, total_sales, discounts GROUP BY sales_channel TIMESERIES month SINCE -6M UNTIL today
FROM sessions SHOW sessions, sessions_with_cart_additions, sessions_that_reached_checkout, sessions_that_completed_checkout SINCE -90d UNTIL today
```

```sql
-- הזמנות שסופקו חסרות, לפי חודש
select date_trunc('month', o.captured_at)::date m, count(distinct o.mirror_id) orders,
       count(distinct o.mirror_id) filter (where ct.mirror_id is not null) short

select item_id, calculated_on_hand from api_read.v_fg_stock_export where item_type='FG' and calculated_on_hand<0 order by 2;

from private_core.v_production_plan_vs_actual
where plan_date between current_date-30 and current_date-1 and status<>'cancelled';
```
